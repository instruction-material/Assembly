cmake_minimum_required(VERSION 3.20)
project(SetupEvidence LANGUAGES C ASM)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(LAB_VALUE 41 CACHE STRING "Integer returned by the owned assembly function")
if(NOT LAB_VALUE MATCHES "^[0-9]+$" OR LAB_VALUE GREATER 255)
    message(FATAL_ERROR "LAB_VALUE must be an integer from 0 through 255")
endif()
set(LAB_ASM_LANGUAGE "assembler-with-cpp" CACHE STRING "Assembly input language")
set_source_files_properties(routine.S PROPERTIES COMPILE_OPTIONS "-x;${LAB_ASM_LANGUAGE}")
enable_testing()
foreach(program IN ITEMS app contracts)
    if(program STREQUAL "app")
        set(caller main.c)
    else()
        set(caller tests.c)
    endif()
    add_executable(${program} ${caller} routine.S)
    target_compile_definitions(${program} PRIVATE LAB_VALUE=${LAB_VALUE})
    target_compile_options(${program} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic;-Werror>")
endforeach()
add_test(NAME configured_function_value COMMAND contracts)
