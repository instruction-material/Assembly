cmake_minimum_required(VERSION 3.20)
project(assembly_hardening LANGUAGES C ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Use Linux x86-64 System V.")
endif()
if(NOT CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang)$")
    message(FATAL_ERROR "Use Clang or GCC.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(LAB_PROFILE "guard-pie" CACHE STRING "baseline, guard, or guard-pie")
set_property(CACHE LAB_PROFILE PROPERTY STRINGS baseline guard guard-pie)
if(LAB_PROFILE STREQUAL "baseline")
    set(lab_c_flags -fno-stack-protector -fno-pie)
    set(lab_link_flags -no-pie)
elseif(LAB_PROFILE STREQUAL "guard")
    set(lab_c_flags -fstack-protector-all -fno-pie)
    set(lab_link_flags -no-pie)
elseif(LAB_PROFILE STREQUAL "guard-pie")
    set(lab_c_flags -fstack-protector-all -fPIE)
    set(lab_link_flags -pie)
else()
    message(FATAL_ERROR "Unknown LAB_PROFILE: ${LAB_PROFILE}")
endif()
add_library(checksums STATIC model.c bytes.S)
add_executable(assembly_hardening main.c)
target_link_libraries(assembly_hardening PRIVATE checksums)
add_executable(hardening_contracts tests.c abi_checks.S)
target_link_libraries(hardening_contracts PRIVATE checksums)
foreach(target checksums assembly_hardening hardening_contracts)
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Werror;-fno-omit-frame-pointer;${lab_c_flags}>")
endforeach()
foreach(target assembly_hardening hardening_contracts)
    target_link_options(${target} PRIVATE ${lab_link_flags})
endforeach()
include(CTest)
add_test(NAME hardening_contracts COMMAND hardening_contracts)
