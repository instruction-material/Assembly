cmake_minimum_required(VERSION 3.20)
project(assembly_opening LANGUAGES C ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR
   NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Use a Linux x86-64 lab for this System V project.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(transform STATIC routine.S)
add_executable(assembly_demo main.c)
target_link_libraries(assembly_demo PRIVATE transform)
add_executable(assembly_contracts tests.c abi_checks.S)
target_link_libraries(assembly_contracts PRIVATE transform)
foreach(target assembly_demo assembly_contracts)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endforeach()
include(CTest)
add_test(NAME transform_contracts COMMAND assembly_contracts)
