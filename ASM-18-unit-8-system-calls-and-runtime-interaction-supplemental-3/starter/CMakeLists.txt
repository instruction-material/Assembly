cmake_minimum_required(VERSION 3.20)
project(OwnedSyscallEvidence LANGUAGES C ASM)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
enable_testing()
add_executable(app main.c routine.S)
add_executable(contracts tests.c routine.S abi_checks.S)
foreach(program IN ITEMS app contracts)
    target_compile_options(${program} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Werror>")
endforeach()
add_test(NAME owned_syscall_contracts COMMAND contracts)
