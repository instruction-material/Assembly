cmake_minimum_required(VERSION 3.20)
project(CompilerObservationLab LANGUAGES C ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "Use Linux x86-64 with the System V AMD64 ABI.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION FALSE)
add_executable(app main.c model.c operation.c baseline.c)
add_executable(contracts tests.c model.c abi_checks.S operation.c baseline.c)
foreach(target IN ITEMS app contracts)
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Werror>")
endforeach()
enable_testing()
add_test(NAME compiler_observation_contract COMMAND contracts)
