cmake_minimum_required(VERSION 3.20)
project(assembly_helper LANGUAGES C ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR
   NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Use a Linux x86-64 lab for this System V project.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(scaled_clamp STATIC scale_and_clamp.S helper.c)
add_executable(assembly_helper main.c)
target_link_libraries(assembly_helper PRIVATE scaled_clamp)
add_executable(helper_contracts tests.c abi_checks.S scale_and_clamp.S helper.c)
target_compile_definitions(helper_contracts PRIVATE ASSEMBLY_HELPER_PROBE)
foreach(target scaled_clamp assembly_helper helper_contracts)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endforeach()
include(CTest)
add_test(NAME helper_contracts COMMAND helper_contracts)
