cmake_minimum_required(VERSION 3.20)
project(assembly_stack LANGUAGES C ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Use Linux x86-64 System V.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(stack STATIC routine.S)
add_executable(assembly_stack main.c)
target_link_libraries(assembly_stack PRIVATE stack)
add_executable(stack_contracts tests.c abi_checks.S routine.S)
target_compile_definitions(stack_contracts PRIVATE STACK_TEST_PROBE)
foreach(target assembly_stack stack_contracts)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endforeach()
include(CTest)
add_test(NAME stack_contracts COMMAND stack_contracts)
