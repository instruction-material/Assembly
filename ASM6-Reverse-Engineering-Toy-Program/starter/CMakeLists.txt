cmake_minimum_required(VERSION 3.20)
project(assembly_reverse LANGUAGES C ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR
   NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Use Linux x86-64 System V.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(REVERSE_SANITIZE "Reject undefined behavior in the C reconstruction" ON)
add_library(reverse STATIC mystery.S reconstructed.c)
add_executable(assembly_reverse main.c)
target_link_libraries(assembly_reverse PRIVATE reverse)
add_executable(reverse_contracts tests.c abi_checks.S)
target_link_libraries(reverse_contracts PRIVATE reverse)
foreach(target reverse assembly_reverse reverse_contracts)
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Werror>")
    if(REVERSE_SANITIZE)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-fsanitize=undefined;-fno-sanitize-recover=undefined>")
    endif()
endforeach()
if(REVERSE_SANITIZE)
    target_link_options(assembly_reverse PRIVATE -fsanitize=undefined -fno-sanitize-recover=undefined)
    target_link_options(reverse_contracts PRIVATE -fsanitize=undefined -fno-sanitize-recover=undefined)
endif()
include(CTest)
add_test(NAME reverse_contracts COMMAND reverse_contracts)
