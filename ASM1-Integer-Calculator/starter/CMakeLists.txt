cmake_minimum_required(VERSION 3.20)
project(assembly_calculator LANGUAGES C ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR
   NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Use a Linux x86-64 lab for this System V project.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(calculator STATIC calculator.S)
add_executable(assembly_calculator main.c)
target_link_libraries(assembly_calculator PRIVATE calculator)
add_executable(calculator_contracts tests.c abi_checks.S)
target_link_libraries(calculator_contracts PRIVATE calculator)
foreach(target assembly_calculator calculator_contracts)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endforeach()
include(CTest)
add_test(NAME calculator_contracts COMMAND calculator_contracts)
