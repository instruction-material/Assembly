cmake_minimum_required(VERSION 3.20)
project(assembly_compiler LANGUAGES C ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Use Linux x86-64 System V.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(model STATIC model.c)
add_executable(assembly_compiler main.c)
target_link_libraries(assembly_compiler PRIVATE model)
add_executable(compiler_contracts tests.c abi_checks.S)
target_link_libraries(compiler_contracts PRIVATE model)
foreach(target model assembly_compiler compiler_contracts)
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Werror>")
endforeach()
include(CTest)
add_test(NAME compiler_contracts COMMAND compiler_contracts)
