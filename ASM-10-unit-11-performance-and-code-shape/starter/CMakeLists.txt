cmake_minimum_required(VERSION 3.20)
project(assembly_performance LANGUAGES C ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Use Linux x86-64 System V.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(loops STATIC loops.S)
add_executable(assembly_performance main.c dataset.c)
target_link_libraries(assembly_performance PRIVATE loops)
add_executable(performance_contracts tests.c abi_checks.S)
target_link_libraries(performance_contracts PRIVATE loops)
foreach(target loops assembly_performance performance_contracts)
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Werror>")
endforeach()
include(CTest)
add_test(NAME performance_contracts COMMAND performance_contracts)
