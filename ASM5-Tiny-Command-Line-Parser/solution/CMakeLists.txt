cmake_minimum_required(VERSION 3.20)
project(assembly_parser LANGUAGES C ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR
   NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Use a Linux x86-64 lab for this System V project.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(parser STATIC parser.S)
add_executable(assembly_parser main.c)
target_link_libraries(assembly_parser PRIVATE parser)
add_executable(parser_contracts tests.c abi_checks.S)
target_link_libraries(parser_contracts PRIVATE parser)
foreach(target assembly_parser parser_contracts)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endforeach()
include(CTest)
add_test(NAME parser_contracts COMMAND parser_contracts)
